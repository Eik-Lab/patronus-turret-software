# Deformable DETR — DeepStream nvinfer primary detector config.
#
# Mirrors the RF-DETR config; the differences that matter for Deformable DETR:
#   * Classification is per-class SIGMOID (focal loss), no background class.
#     Handled by our custom parser (deepstream_deformable_detr_bbox).
#   * Output layers are named "pred_logits" and "pred_boxes" (rename in the
#     parser if your ONNX export uses different names).
#   * Set-prediction needs no NMS/clustering -> cluster-mode=4.
#
# The image normalization is the DETR/ImageNet standard, identical to RF-DETR:
#   means = [0.485, 0.456, 0.406], stds = [0.229, 0.224, 0.225]
# giving net-scale-factor = 1 / (mean_std * 255) = 1 / (0.226 * 255)
# and offsets = means * 255.

[property]
# Deformable DETR (COCO) is trained with 91 classes. Unlike RF-DETR there is no
# dedicated background index; after fine-tuning for drones set this to your
# class count (e.g. 1) and update labelfile-path accordingly.
num-detected-classes=91

# y = net-scale-factor * (x - offset)
net-scale-factor=0.0173520736
offsets=123.675;116.28;103.53

# ONNX must be exported at a fixed input size (e.g. 640x640) so TensorRT can
# build a static engine. The .engine is generated on first run from the ONNX.
onnx-file=../../models/deformable-detr.onnx
model-engine-file=../../models/deformable-detr.onnx_b1_gpu0_fp32.engine

batch-size=1
# 0 = FP32, 1 = INT8, 2 = FP16. Start at FP32; switch to 2 once verified.
network-mode=0
# 0 = RGB (Deformable DETR trains on RGB)
model-color-format=0
# 0 = NCHW
network-input-order=0
network-type=0
gie-unique-id=1
process-mode=1

# Preserve aspect ratio with symmetric padding, matching DETR preprocessing.
maintain-aspect-ratio=1
symmetric-padding=1

labelfile-path=coco91_labels.txt

# Custom sigmoid-based bbox parser (built from deps/deepstream-deformable-detr).
parse-bbox-func-name=deepstream_deformable_detr_bbox
custom-lib-path=../../deps/deepstream-deformable-detr/libdeepstream-deformable-detr.so

# Set-prediction: no clustering stage.
cluster-mode=4

# DLA untested for this model.
enable-dla=0

[class-attrs-all]
# Confidence threshold applied inside the custom parser (sigmoid score).
pre-cluster-threshold=0.5
roi-top-offset=0
roi-bottom-offset=0
